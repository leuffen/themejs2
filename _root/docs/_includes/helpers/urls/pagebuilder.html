{%- comment -%} Returns the page-builder URL for a page. Required: - path: page source path - lang: page language Optional: - base_url: page-builder base URL; defaults to site.data.general.pagebuilder {%- endcomment -%} {%- assign requested_path = include.path -%} {%- assign requested_lang = include.lang -%} {%- assign base_url = include.base_url | default: site.data.general.pagebuilder -%} {%- if base_url and requested_path and requested_lang -%} {{- base_url -}}/page?path={{ requested_path | url_encode }}&lang={{ requested_lang | url_encode }} {%- else -%} # {%- endif -%}
